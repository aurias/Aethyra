gale-1,14,22,0|script|Windkeeper Ama|105
{
    mes "[Windkeeper Ama]";
    mes "\"Welcome to the meadow. Feel how the wind leans on the grass? That is Gale.\"";
    next;
    menu "How do I use Gale?", L_Skills,
         "Can you make something from my harvest?", L_Craft,
         "Goodbye.", L_Close;

L_Skills:
    mes "[Windkeeper Ama]";
    mes "\"Face the way you want to go, then:\"";
    mes "X - Dash: rush forward on the wind.";
    mes "C - Gust: blow Gust Hoppers away from you.";
    mes "V - Wind Scythe: cut grass, herbs and flowers to harvest them.";
    next;
    mes "[Windkeeper Ama]";
    mes "\"Each one draws on your Gale energy, the green bar. It returns quickly while you breathe the open air.\"";
    mes "\"Breeze Reeds hold a little wind. Choose them as supply in your pack (Supply button) and they lend you the current you cannot channel alone. Ask too much of them and they burst.\"";
    mes "\"Bring me three Meadow Herbs and three Windflower Petals and I will brew you a Gale Tonic.\"";
    goto L_Close;

L_Craft:
    if (countitem(703) < 3 || countitem(704) < 3)
        goto L_Short;
    delitem 703, 3;
    delitem 704, 3;
    getitem 710, 1;
    mes "[Windkeeper Ama]";
    mes "\"There. A Gale Tonic: it mends wounds and fills your lungs with wind.\"";
    goto L_Close;

L_Short:
    mes "[Windkeeper Ama]";
    mes "\"I need three Meadow Herbs from the terrace above and three Windflower Petals from by the pond.\"";
    goto L_Close;

L_Close:
    close;
}

// Wren keeps the terrace above the stairs and teaches its ways down and up.
gale-1,11,7,0|script|Wren the Ridge-runner|106
{
    mes "[Wren the Ridge-runner]";
    mes "\"Stairs are for carrying things. The wind is for getting places.\"";
    next;
    menu "Teach me to move on the wind.", L_Teach,
         "Where is it worth going?", L_Where,
         "Goodbye.", L_Close;

L_Teach:
    set @taught, huegrant(4, 1) + huegrant(5, 1);
    mes "[Wren the Ridge-runner]";
    if (@taught)
        mes "\"Step to a cliff edge and face it. F lets the wind carry you down - Featherfall. G throws you up a ledge - Upward Jump.\"";
    if (@taught == 0)
        mes "\"You know the moves already. Practise.\"";
    mes "\"Watch the ground ahead of you: a marker shows where you would land, or why you can't.\"";
    mes "\"A narrow ledge you can jump. The terrace face here is too wide until you have practised.\"";
    goto L_Close;

L_Where:
    mes "[Wren the Ridge-runner]";
    mes "\"Up on the lookout, east of here, Skyreeds grow where nobody without wind can reach. Their stems are the best Breeze Reeds.\"";
    mes "\"Hoppers love the terrace edge. A Gust at the right moment sends them tumbling to the meadow.\"";
    goto L_Close;

L_Close:
    close;
}
